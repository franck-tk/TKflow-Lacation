const checkCNI = async (req, res, next) => {
  const user = req.user;
  if (user.email === 'todjoumfranck@gmail.com' || user.role === 'admin' || user.role === 'superadmin') {
    return next();
  }
  if (!user.cni_number ||!user.cni_photo_url) {
    return res.status(403).json({error: 'CNI_REQUIRED', message: 'Upload your CNI first'});
  }
  next();
};
module.exports = checkCNI;

import { useState } from 'react';
export default function CNIUpload({ onUpload }) {
  const [preview, setPreview] = useState(null);
  
  const handleFile = (e) => {
    const file = e.target.files[0];
    if (file) {
      setPreview(URL.createObjectURL(file));
      onUpload(file);
    }
  };

  return (
    <div className="bg-gray-800 p-6 rounded-lg mb-6">
      <h3 className="text-white text-xl mb-4">Photo de votre CNI</h3>
      <div className="grid grid-cols-2 gap-4 mb-4">
        <label className="bg-orange-500 text-white p-4 rounded-lg text-center cursor-pointer hover:bg-orange-600">
          <input type="file" accept="image/*" onChange={handleFile} className="hidden"/>
          📁 Choisir Galerie
        </label>
        <label className="bg-orange-500 text-white p-4 rounded-lg text-center cursor-pointer hover:bg-orange-600">
          <input type="file" accept="image/*" capture="environment" onChange={handleFile} className="hidden"/>
          📷 Prendre Photo
        </label>
      </div>
      {preview && <img src={preview} alt="CNI Preview" className="w-full rounded-lg"/>}
    </div>
  );
}
